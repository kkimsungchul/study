# rocky linux에 Fluent Bit 설치

# 사용 이유
	오픈텔레메트리 수집로그 -> 미정 -> 카프카 -> 아파치 스톰
	위 순서대로 데이터를 보내서 처리하려고함
	다만 오픈텔레메트리 로그규격에서 원하는 데이터를 찾으려면 4뎁스까지 내려가야함
	이걸 아파치 스톰에서 처리하기에는 너무 부하가 커서 전처리 단계를 추가하기 위함
	dataprepper 은 테스트 완료하였으나, 속도측면에서 Fluent Bit가 더 좋다고 하여 테스트 하기위해 설치함

# 공식 가이드
	https://docs.fluentbit.io/manual

# 설치 가이드
	https://docs.fluentbit.io/manual/installation/downloads/linux/alma-rocky





# 저장소 추가
	※ root로 실행

	- 저장소 추가
	=====================================================================
	cd /etc/yum.repos.d
	vi fluent-bit.repo
	=====================================================================

	- 추가 할 내용
	=====================================================================
	[fluent-bit]
	name=Fluent Bit
	baseurl=https://packages.fluentbit.io/almalinux/$releasever/
	gpgcheck=1
	gpgkey=https://packages.fluentbit.io/fluentbit.key
	repo_gpgcheck=1
	enabled=1
	=====================================================================

# 설정파일 수정
	위치 : /etc/fluent-bit/fluent-bit.conf
	
	- 기존
	=====================================================================
	[INPUT]
		name cpu
		tag  cpu.local

		# Read interval (sec) Default: 1
		interval_sec 1

	[OUTPUT]
		name  stdout
		match *

	=====================================================================
	
	- 변경
	※ 탭들어가면 안되고 스페이스 4칸임
	=====================================================================
	[SERVICE]
		Flush         1
		Log_Level     info
		Daemon        off
		Parsers_File  parsers.conf

	[INPUT]
		Name          opentelemetry
		Listen        0.0.0.0
		Port          4318

	[OUTPUT]
		Name          file
		Match         *
		Path          /home/sungchul/fluent-bit-log
		File          otel-logs.log
	=====================================================================

	- 오류 수정
		아래파일에서 //etc 이부분을 /etc로 수정
		왜 이렇게 됐는지 모르겠음. 나는 yum 으로 설치만 한게 다임
	=====================================================================
	 vi /usr/lib/systemd/system/fluent-bit.service

	[Unit]
	Description=Fluent Bit
	Documentation=https://docs.fluentbit.io/manual/
	Requires=network.target
	After=network.target

	[Service]
	Type=simple
	EnvironmentFile=-/etc/sysconfig/fluent-bit
	EnvironmentFile=-/etc/default/fluent-bit
	ExecStart=/opt/fluent-bit/bin/fluent-bit -c //etc/fluent-bit/fluent-bit.conf
	Restart=always

	[Install]
	WantedBy=multi-user.target
	=====================================================================

	- systemed 리로드 
	=====================================================================
	systemctl daemon-reload
