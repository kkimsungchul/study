# OTel - Opentelemetry config 설정


# 127.0.0.1과 0.0.0.0의 차이점

	- 127.0.0.1 (Loopback Address):
		127.0.0.1은 로컬 컴퓨터 내에서만 접근할 수 있는 루프백(Loopback) 주소입니다.
		서버가 이 주소에 바인딩되면, 같은 서버 내에서만 접근이 가능합니다. 외부에서 요청이 와도 접속이 차단됩니다.
		예를 들어 localhost:9464로 설정된 서비스를 외부에서 접근할 수 없는 이유가 여기에 있습니다.

	- 0.0.0.0 (Wildcard Address):
		0.0.0.0은 와일드카드 주소로, 서버의 모든 네트워크 인터페이스에서 접근할 수 있게 합니다.
		서버가 이 주소에 바인딩되면, 로컬 네트워크뿐 아니라 외부 네트워크에서도 접근이 가능합니다.
		예를 들어 0.0.0.0:9464로 설정하면, 같은 서버뿐 아니라 다른 네트워크의 클라이언트도 서버 IP:9464로 접근할 수 있습니다.
		따라서, 외부에서 접속하려면 127.0.0.1 대신 0.0.0.0으로 설정해야 합니다.

	※ 외부에서도 접근 또는 수집을 가능하도록 하게 하려면 0.0.0.0을 사용
	※ 외부에서 접근할 필요가 없는 경우 127.0.0.1 또는 localhost 사용

# contrib 사용하여 여러 기능 사용하기
	- 아래처럼 사용
	- log_statements 에서 로그에 들어있는 데이터 파싱 가능
	- 속성들의 위치를 추가할수 있음
	=====================================================================
	receivers:
	  otlp:
		protocols:
		  grpc:
			endpoint: 127.0.0.1:9999




	#묶어서 보내지 않도록 설정
	#다만 해도 묶어서 보냄..
	processors:
	  batch:
		timeout: 1ms
		send_batch_size: 1

	# traceid와 spanid 를 속성에 매핑
	  # logRecords flatten
	  transform:
		log_statements:
		  - context: log
			statements:
			  # 1body.stringValue → body 로 단순화
			  - set(body, body.string) where body.string != nil

			  # 2trace / span 승격
			  - set(attributes["trace_id"], trace_id.string)
			  - set(attributes["span_id"], span_id.string)

			  # 3TOPIC_LAMP → attributes (Kafka routing용)
			  - set(attributes["topic"], resource.attributes["TOPIC_LAMP"])

			  # 4OTEL 메타 제거
			  - delete_key(resource.attributes, "*")
			  - delete_key(attributes, "otel.*")


	  transform/logs:
		log_statements:
		  - context: log
			statements:
			  # 1️ JSON 문자열만 파싱 (첫 글자가 '{')
			  - set(body, ParseJSON(body.string))
				where IsString(body)
				  and Substring(body.string, 0, 1) == "{"

			  # 2️ trace / span 주입 (JSON 로그만)
			  - set(body["trace_id"], trace_id.string)
				where IsMap(body)

			  - set(body["span_id"], span_id.string)
				where IsMap(body)

			  # 3️ topic 주입
			  - set(body["topic"], resource.attributes["TOPIC_LAMP"])
				where IsMap(body)
			  # 4 service_name 주입
			  - set(body["service_name"], resource.attributes["service.name"])
				where IsMap(body)

			  # 5 response 파싱(json)
			  - set(body["response"], ParseJSON(body["response"]))
				where IsMap(body)
				  and IsString(body["response"])
				  and Substring(body["response"], 0, 1) == "{"

			  # 6 security 파싱
			  - set(body["security"], ParseJSON(body["security"]))
				where IsMap(body)
				  and IsString(body["security"])
				  and Substring(body["security"], 0, 1) == "{"

			  # 7 security.detail 파싱
			  - set(body["security"]["detail"], ParseJSON(body["security"]["detail"]))
				where IsMap(body)
				  and IsString(body["security"]["detail"])
				  and Substring(body["security"]["detail"], 0, 1) == "{"
	 
			  # 8 host 파싱
			  - set(body["host"], ParseJSON(body["host"]))
				where IsMap(body)
				  and IsString(body["host"])
				  and Substring(body["host"], 0, 1) == "{"

	exporters:
	  # =====================
	  # FILE EXPORTERS (분리)
	  # =====================
	  file/logs:
		path: logs.json

	  file/traces:
		path: traces.json

	  file/metrics:
		path: metrics.json

	  # =====================
	  # OTHER EXPORTERS

		verbosity: detailed
