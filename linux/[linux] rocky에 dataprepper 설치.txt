# rocky linux에 데이터 프레퍼(Data Prepper) 설치

# 사용 이유
	오픈텔레메트리 콜렉터에서 수집한 로그들을 보기편하게 파싱하려고 사용함
	이거 윈도우 버전이 옛날엔 있었는지 모르겠는데, 이제는 리눅스랑 도커만 제공해주고있음


# 계정 생성
	서비스에 등록하지 않고 계정으로 직접 실행할예정임
	=====================================================================
	useradd dataprepper
	passwd dataprepper
	=====================================================================

# 계정 전환
	=====================================================================
	su - dataprepper
	=====================================================================

	=====================================================================

# 압축 해제
	=====================================================================
	tar -xzf opensearch-data-prepper-jdk-2.13.0-linux-x64.tar.gz
	=====================================================================

# 데이터 프레퍼 파일경로
	- 실행파일 
	=====================================================================
	/home/dataprepper/opensearch-data-prepper-jdk-2.13.0-linux-x64/bin/data-prepper
	=====================================================================
	
	- 설정파일
	=====================================================================
	/home/dataprepper/opensearch-data-prepper-jdk-2.13.0-linux-x64/config/data-prepper-config.yaml
	=====================================================================

# 설정
	나는 오픈텔레메트리 콜렉터에서 보내는 로그를 받아 분리하는 작업을 할 예정임
	!!주의사항!!
	설정파일내에 tap이 들어가잇으면 오류남
	=====================================================================
	Caused by: while scanning for the next token
	found character '\t(TAB)' that cannot start any token. (Do not use \t(TAB) for indentation)
	 in 'reader', line 3, column 1:
			otel_logs_source:
	=====================================================================

	- 기존 파일 백업
	=====================================================================
	cp data-prepper-config.yaml data-prepper-config.yaml.2026.01.22
	=====================================================================
	
	- data-prepper-config.yaml
	=====================================================================
	ssl: false

	server_port : 4900

	=====================================================================
	
	- pipelines.yaml
	=====================================================================
	lamp-log-pipeline:
	  source:
		otel_logs_source:
		  port: 21890  # gRPC로 받을 포트 설정
		  ssl: false   # 필요에 따라 SSL 설정

	  sink:
		- file:
			path: /home/dataprepper/log/lamp-logs.json
			append: true

	=====================================================================

# 로그 디렉토리 생성
	※ dataprepper 계정으로 진행
	※ 혹시나 root나 뭐 다른걸로 했으면 권한,소유자는 dataprepper로 변경하자
	=====================================================================
	mkdir -p /home/dataprepper/log
	sudo chown -R $(whoami):$(whoami) /home/dataprepper/log
	=====================================================================

# 데이터 프레퍼 실행
	
	=====================================================================
	cd /home/dataprepper/opensearch-data-prepper-jdk-2.13.0-linux-x64/

	./bin/data-prepper config/pipelines.yaml config/data-prepper-config.yaml
	=====================================================================

# 방화벽 오픈
	=====================================================================
	firewall-cmd --add-port=21890/tcp --permanent
	firewall-cmd --reload
	=====================================================================

# 정상 오픈 확인
	다른서버에서 접속
	=====================================================================
	telnet 아이피 21890
	=====================================================================
